function actualizarBarras(datos) { console.log("Registros recibidos:", datos.length); const canvas = document.getElementById("graficaAreas"); if (!canvas) { console.error("No se encontró graficaAreas"); estado("Error: no existe el contenedor de la gráfica."); return; } if (typeof Chart === "undefined") { console.error("Chart.js no está disponible"); estado("Error al cargar Chart.js."); return; } const mayores = datos .map(a => ({ ...a, AREA_NUM: valorNumerico(a.AREA_N_KM2) })) .filter(a => a.AREA_NUM !== null && Number.isFinite(a.AREA_NUM) && a.AREA_NUM >= 0 ) .sort((a, b) => b.AREA_NUM - a.AREA_NUM) .slice(0, 10); if (mayores.length === 0) { estado("No se encontraron áreas válidas."); return; } if (graficaAreas) { graficaAreas.destroy(); graficaAreas = null; } graficaAreas = new Chart( canvas.getContext("2d"), { type: "bar", data: { labels: mayores.map(a => a.NMGCUENCA || "Cuenca sin nombre" ), datasets: [{ label: "Área natural (km²)", data: mayores.map(a => a.AREA_NUM), backgroundColor: "#3386a0", borderColor: "#176181", borderWidth: 1, borderRadius: 4 }] }, options: { indexAxis: "y", responsive: true, maintainAspectRatio: false, plugins: { legend: { display: false }, tooltip: { callbacks: { label: ctx => "Área natural: " + numero(ctx.raw) + " km²" } } }, scales: { x: { beginAtZero: true, title: { display: true, text: "Área natural (km²)" } }, y: { ticks: { font: { size: 10 } } } }, onClick: (evento, elementos) => { if (!elementos.length) return; const posicion = elementos[0].index; const cuenca = mayores[posicion]; if (cuenca) { seleccionarPorId(cuenca); } } } } ); console.log( "Gráfica generada correctamente:", mayores.length, "cuencas" ); }
